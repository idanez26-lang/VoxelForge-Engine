# Tests VoxelForge — déclaration compacte.
# Ce fichier est une réécriture 1:1 de l'ancien tests/CMakeLists.txt :
# mêmes cibles, mêmes sources, mêmes libs, mêmes noms de tests, mêmes
# propriétés CTest. Vérifié par comparaison sémantique automatique.

function(vf_configure_test target)
    target_compile_features(${target} PRIVATE cxx_std_20)

    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive-)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endfunction()

# Crée l'exécutable de test et le configure.
#   vf_test_target(<target> SOURCES <src...> [LINK <libs...>] [EDITOR_INCLUDES])
function(vf_test_target target)
    cmake_parse_arguments(ARG "EDITOR_INCLUDES" "" "SOURCES;LINK" ${ARGN})
    add_executable(${target} ${ARG_SOURCES})
    if(ARG_EDITOR_INCLUDES)
        target_include_directories(${target} PRIVATE ${PROJECT_SOURCE_DIR}/editor/src)
    endif()
    if(ARG_LINK)
        target_link_libraries(${target} PRIVATE ${ARG_LINK})
    endif()
    vf_configure_test(${target})
endfunction()

# Enregistre un test CTest sur une cible existante.
#   vf_test_case(<nom.ctest> <target> [ARGS <args...>] [LABELS <l...>] [TIMEOUT <s>])
function(vf_test_case name target)
    cmake_parse_arguments(ARG "" "TIMEOUT" "ARGS;LABELS" ${ARGN})
    add_test(NAME ${name} COMMAND $<TARGET_FILE:${target}> ${ARG_ARGS})
    if(ARG_LABELS)
        set_tests_properties(${name} PROPERTIES LABELS "${ARG_LABELS}")
    endif()
    if(ARG_TIMEOUT)
        set_tests_properties(${name} PROPERTIES TIMEOUT ${ARG_TIMEOUT})
    endif()
endfunction()

# Smoke test de l'éditeur complet (exécute VoxelForgeEditor avec un drapeau).
function(vf_editor_smoke name flag)
    add_test(NAME ${name} COMMAND $<TARGET_FILE:VoxelForgeEditor> ${flag})
    set_tests_properties(
        ${name}
        PROPERTIES
            TIMEOUT 30
            WORKING_DIRECTORY ${PROJECT_SOURCE_DIR}
            ENVIRONMENT
                "VOXELFORGE_RECENT_PROJECTS_FILE=${CMAKE_CURRENT_BINARY_DIR}/smoke-recent-projects.txt;VOXELFORGE_PREFERENCES_FILE=${CMAKE_CURRENT_BINARY_DIR}/smoke-preferences.ini"
    )
endfunction()
